# Get the highest intensity 5'boundary spot (channel 3) per cell
ch3_spots_selected <- spot_filt_tbl %>%
  filter(channel == 3) %>%
  group_by(Experiment, cell_index) %>%
  slice_max(integrated_intensity, n = 1, with_ties = FALSE) %>%
  ungroup() %>%
  select(Experiment, cell_index, ch3_x = x, ch3_y = y, ch3_intensity = integrated_intensity)

ch2_spots <- spot_filt_tbl %>%
  filter(channel == 2) %>%
  select(Experiment, cell_index, spot_index, ch2_x = x, ch2_y = y, ch2_intensity = integrated_intensity)

# Calculate distance between 5' spot and all 3' spots in the same cell
ch2_ch3_distances <- ch3_spots_selected %>%
  left_join(ch2_spots, 
            by = c("Experiment", "cell_index"),
            relationship = "many-to-many") %>%
  mutate(ch2_ch3_distance = sqrt((ch2_x - ch3_x)^2 + (ch2_y - ch3_y)^2))

# Get the closest 3' spot for each 5' spot
ch2_ch3_closest <- ch2_ch3_distances %>%
  group_by(Experiment, cell_index) %>%
  slice_min(ch2_ch3_distance, n = 1, with_ties = FALSE) %>%
  ungroup() %>%
  select(Experiment, cell_index, closest_ch2_spot = spot_index, 
         ch2_ch3_distance, ch2_x, ch2_y, ch2_intensity, ch3_x, ch3_y, ch3_intensity)

Classify cells based on 5’-3’ boundary distance

ch2_ch3_classified <- ch2_ch3_closest %>%
  mutate(ch3_proximity = case_when(
    ch2_ch3_distance < THRESHOLD1_CH2_CH3 ~ "Proximal",
    ch2_ch3_distance >= THRESHOLD1_CH2_CH3 & ch2_ch3_distance < THRESHOLD2_CH2_CH3 ~ "Distal",
    ch2_ch3_distance >= THRESHOLD2_CH2_CH3 ~ "No 3' spot",
    TRUE ~ NA_character_
  ))

# Get RNA spots
ch1_spots <- spot_filt_tbl %>%
  filter(channel == 1) %>%
  select(Experiment, cell_index, ch1_spot_index = spot_index, ch1_x = x, ch1_y = y, ch1_intensity = integrated_intensity)

# Calculate distance between 5'boundary spot and all RNA spots in the same cell
ch3_ch1_distances <- ch2_ch3_classified %>%
  left_join(ch1_spots, 
            by = c("Experiment", "cell_index"),
            relationship = "many-to-many") %>%
  mutate(ch3_ch1_distance = sqrt((ch3_x - ch1_x)^2 + (ch3_y - ch1_y)^2))

# Get the closest RNA spot for each 5' spot
ch3_ch1_closest <- ch3_ch1_distances %>%
  group_by(Experiment, cell_index) %>%
  slice_min(ch3_ch1_distance, n = 1, with_ties = FALSE) %>%
  ungroup()

# Classify alleles as Active/Inactive
spots_classified <- ch2_ch3_classified %>%
  left_join(
    ch3_ch1_closest %>% 
      select(Experiment, cell_index,
             closest_ch1_spot = ch1_spot_index, ch3_ch1_distance, 
             ch1_x, ch1_y, ch1_intensity),
    by = c("Experiment", "cell_index")
  ) %>%
  mutate(
    # Replace ch1_int with NA when ch1_int is too low or ch1 spot is too far
   ch1_intensity = if_else(
      ch1_intensity < THRESHOLD4_CH1_INT | ch3_ch1_distance > THRESHOLD3_CH3_CH1,
      NA_real_,
      ch1_intensity
    ),
    # Classify using the updated ch1_intensity
    activity_status = case_when(
  !is.na(ch1_intensity) ~ "Active",
  TRUE ~ "Inactive"
    )
  )

# Extract experiment name
spots_classified <- spots_classified %>%
  mutate(experiment_name = str_remove(Experiment, "_\\d+\\.czi$"))
# Filter out cells with "No 3' spot" classification
spots_classified <- spots_classified %>%
  filter(ch3_proximity != "No 3' spot")

# Save table with individual spot information
write_csv(spots_classified, "individual_allele_data.csv")

Calculate fraction of active/inactive alleles according to 5’-3’ proximity status

proximity_activity_summary <- spots_classified %>%
  group_by(experiment_name, ch3_proximity, activity_status) %>%
  summarise(
    n_cells = n(),
    mean_ch1_intensity_active = if_else(
      first(activity_status) == "Active",
      mean(ch1_intensity, na.rm = TRUE),
      NA_real_
    ),
    median_ch1_intensity_active = if_else(
      first(activity_status) == "Active",
      median(ch1_intensity, na.rm = TRUE),
      NA_real_
    ),
    .groups = "drop"
  ) %>%
  group_by(experiment_name, ch3_proximity) %>%
  mutate(
    total_cells = sum(n_cells),
    fraction = n_cells / total_cells
  ) %>%
  ungroup()

# Save to CSV file
write_csv(proximity_activity_summary, "proximity_activity_summary.csv")
